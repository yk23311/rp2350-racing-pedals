cmake_minimum_required(VERSION 3.21)
option(PEDAL_FIRMWARE "Build RP2350 firmware instead of native tests" OFF)
option(PEDAL_USB_XINPUT "Use native XInput RT/LT rather than legacy HID axes" ON)
option(PEDAL_USB_HOST "Build the USB-A host probe path (requires PEDAL_USB_XINPUT)" OFF)
if(PEDAL_FIRMWARE)
  if(NOT PICO_SDK_PATH AND DEFINED ENV{PICO_SDK_PATH})
    set(PICO_SDK_PATH "$ENV{PICO_SDK_PATH}")
  endif()
  if(NOT EXISTS "${PICO_SDK_PATH}/pico_sdk_init.cmake")
    message(FATAL_ERROR "Set PICO_SDK_PATH to Pico SDK 2.2.0 or newer")
  endif()
  set(PICO_BOARD waveshare_rp2350_usb_a CACHE STRING "Target board")
  if(NOT PICO_BOARD STREQUAL "waveshare_rp2350_usb_a")
    message(FATAL_ERROR "This firmware layout is only verified for waveshare_rp2350_usb_a")
  endif()
  if(PEDAL_USB_HOST AND NOT PEDAL_USB_XINPUT)
    message(FATAL_ERROR "PEDAL_USB_HOST requires PEDAL_USB_XINPUT=ON")
  endif()
  if(PEDAL_USB_HOST)
    set(PICO_PIO_USB_PATH "${CMAKE_CURRENT_LIST_DIR}/deps/Pico-PIO-USB" CACHE PATH "Pico-PIO-USB source directory")
  endif()
  include("${PICO_SDK_PATH}/pico_sdk_init.cmake")
  project(pedals C CXX ASM)
  pico_sdk_init()
  if(PICO_SDK_VERSION_STRING VERSION_LESS "2.2.0")
    message(FATAL_ERROR "Pico SDK 2.2.0 or newer is required")
  endif()
else()
  project(pedals C)
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(pedal_core src/core/pedal.c src/core/config_record.c src/core/console.c src/core/report_gate.c src/core/usb_layout.c src/core/xinput_protocol.c src/core/xinput_pad.c src/core/bridge.c)
target_compile_definitions(pedal_core PUBLIC PEDAL_USB_XINPUT=$<BOOL:${PEDAL_USB_XINPUT}>)
target_include_directories(pedal_core PUBLIC include)
if(MSVC)
  target_compile_options(pedal_core PRIVATE /W4 /WX)
else()
  target_compile_options(pedal_core PRIVATE -Wall -Wextra -Werror)
  target_link_libraries(pedal_core PUBLIC m)
endif()
if(PEDAL_FIRMWARE)
  add_subdirectory(firmware)
else()
  enable_testing()
  add_executable(test_core tests/test_core.c)
  target_link_libraries(test_core PRIVATE pedal_core)
  foreach(case defaults mapping filtering hysteresis curve calibration modes invalid button)
    add_test(NAME core_${case} COMMAND test_core ${case})
  endforeach()
  add_executable(test_config tests/test_config.c)
  target_link_libraries(test_config PRIVATE pedal_core)
  add_test(NAME config_records COMMAND test_config)
  add_executable(test_console tests/test_console.c)
  target_link_libraries(test_console PRIVATE pedal_core)
  add_test(NAME console_protocol COMMAND test_console)
  add_executable(test_usb tests/test_usb.c)
  target_link_libraries(test_usb PRIVATE pedal_core)
  add_test(NAME usb_contract COMMAND test_usb)
  add_executable(test_storage tests/test_storage.c firmware/storage.c)
  target_include_directories(test_storage PRIVATE tests/hal_mock firmware)
  target_link_libraries(test_storage PRIVATE pedal_core)
  add_test(NAME flash_power_loss COMMAND test_storage)
  add_executable(test_xinput tests/test_xinput.c)
  target_link_libraries(test_xinput PRIVATE pedal_core)
  foreach(case packet layout control)
    add_test(NAME xinput_${case} COMMAND test_xinput ${case})
  endforeach()
  add_executable(test_xinput_pad tests/test_xinput_pad.c)
  target_link_libraries(test_xinput_pad PRIVATE pedal_core)
  foreach(case parse neutral predicates)
    add_test(NAME xinput_pad_${case} COMMAND test_xinput_pad ${case})
  endforeach()
  add_executable(test_bridge tests/test_bridge.c)
  target_link_libraries(test_bridge PRIVATE pedal_core)
  foreach(case pedal_only handshake override max timeout unmount blocked quantize)
    add_test(NAME bridge_${case} COMMAND test_bridge ${case})
  endforeach()
  add_executable(test_xinput_driver tests/test_xinput_driver.c firmware/xinput_device.c)
  target_include_directories(test_xinput_driver PRIVATE tests/usb_mock firmware)
  target_link_libraries(test_xinput_driver PRIVATE pedal_core)
  add_test(NAME xinput_driver COMMAND test_xinput_driver)
  if(PEDAL_USB_XINPUT)
    add_executable(test_native_usb tests/test_native_usb.c firmware/usb.c)
    target_include_directories(test_native_usb PRIVATE tests/usb_mock firmware)
    target_link_libraries(test_native_usb PRIVATE pedal_core)
    add_test(NAME xinput_usb_integration COMMAND test_native_usb)
  endif()
endif()
