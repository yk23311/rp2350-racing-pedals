add_executable(pedals main.c hal.c storage.c usb.c)
if(PEDAL_USB_XINPUT)
  target_sources(pedals PRIVATE xinput_device.c)
endif()
target_include_directories(pedals PRIVATE ${CMAKE_CURRENT_LIST_DIR})
target_link_libraries(pedals PRIVATE pedal_core pico_stdlib hardware_adc hardware_flash pico_flash pico_unique_id tinyusb_device)
target_compile_options(pedals PRIVATE -Wall -Wextra -Werror)
pico_enable_stdio_usb(pedals 0)
pico_enable_stdio_uart(pedals 0)
# Use the SDK layout, reserving two entire sectors without changing physical flash size.
file(READ "${PICO_SDK_PATH}/src/rp2_common/pico_crt0/rp2350/memmap_default.ld" PEDAL_LINKER)
string(REPLACE "INCLUDE \"pico_flash_region.ld\"" "FLASH(rx) : ORIGIN = 0x10000000, LENGTH = 0x1fe000" PEDAL_LINKER "${PEDAL_LINKER}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/pedals.ld" "${PEDAL_LINKER}")
pico_set_linker_script(pedals "${CMAKE_CURRENT_BINARY_DIR}/pedals.ld")
pico_set_program_name(pedals "RP2350 Dual Pedals")
if(PEDAL_USB_XINPUT)
  pico_set_program_version(pedals "0.2.0-xinput")
else()
  pico_set_program_version(pedals "0.1.0-hid")
endif()
pico_add_extra_outputs(pedals)
